A-D versus the common Control

In the additive models, heat_group coefficients are adjusted A-D minus Control differences, assumed common across years and cultivars. Weights are in grams; healthy percentage differences are in percentage points. Intervals are pointwise. Holm p values are adjusted across the four contrasts within each response.

models <- list(total_weight=m_total, healthy_weight=m_healthy,
               healthy_count_pct=m_count_pct, healthy_weight_pct=m_weight_pct)
effects <- do.call(rbind, lapply(names(models), function(name) {
  fit <- models[[name]]
  tab <- summary(fit)$tTable
  tab <- tab[grepl("^heat_group", rownames(tab)), , drop=FALSE]
  data.frame(outcome=name,
    comparison=paste(sub("heat_group", "", rownames(tab)), "- Control"),
    estimate=tab[, "Value"],
    lower=tab[, "Value"] - qt(.975, tab[, "DF"]) * tab[, "Std.Error"],
    upper=tab[, "Value"] + qt(.975, tab[, "DF"]) * tab[, "Std.Error"],
    p_value=tab[, "p-value"],
    p_holm=p.adjust(tab[, "p-value"], method="holm"), row.names=NULL)
}))

Intensity comparisons: A-D

A, B, C and D correspond to 4, 3, 2 and 1 heat pulses. Weight outcomes use the cultivar-interaction models; percentage outcomes use additive models. Means are averaged equally over years (and cultivars for percentage outcomes). Controls remain pooled within each year and cultivar. Comparisons use Holm correction for the six A-D pairs within each outcome/cultivar; confidence intervals are pointwise 95%, not simultaneous. Degrees of freedom use a conservative minimum over the fixed coefficients involved.

# No additional packages are required.
intensity_results <- function(fit, cultivar_level=NULL) {
  grid <- expand.grid(
    year=levels(data_clean$year),
    cultivar=if (is.null(cultivar_level)) levels(data_clean$cultivar) else cultivar_level,
    heat_group=levels(data_clean$heat_group)
  )
  grid$year <- factor(grid$year, levels=levels(data_clean$year))
  grid$cultivar <- factor(grid$cultivar, levels=levels(data_clean$cultivar))
  grid$heat_group <- factor(grid$heat_group, levels=levels(data_clean$heat_group))
  X <- model.matrix(delete.response(terms(formula(fit))), grid)
  X <- X[, names(fixef(fit)), drop=FALSE]
  mean_X <- do.call(rbind, lapply(levels(data_clean$heat_group), function(g) {
    colMeans(X[grid$heat_group == g, , drop=FALSE])
  }))
  rownames(mean_X) <- levels(data_clean$heat_group)

  estimate_contrast <- function(L) {
    estimate <- as.numeric(L %*% fixef(fit))
    se <- sqrt(as.numeric(L %*% vcov(fit) %*% L))
    df <- min(fit$fixDF$X[abs(L) > 1e-10])
    data.frame(estimate=estimate, SE=se, df=df,
      lower=estimate-qt(.975,df)*se,
      upper=estimate+qt(.975,df)*se,
      p_value=2*pt(-abs(estimate/se),df))
  }
  means <- do.call(rbind, lapply(rownames(mean_X), function(g) {
    cbind(group=g, estimate_contrast(mean_X[g, ]))
  }))
  # Pairwise comparisons ONLY among heated groups.
  pairs <- combn(c("A", "B", "C", "D"), 2, simplify=FALSE)
  comparisons <- do.call(rbind, lapply(pairs, function(g) {
    cbind(comparison=paste(g, collapse=" - "),
      estimate_contrast(mean_X[g[1], ] - mean_X[g[2], ]))
  }))
  comparisons$p_holm <- p.adjust(comparisons$p_value, method="holm")
  list(means=means, comparisons=comparisons)
}

# A-D scheme effects relative to the common Control can also be compared
# by subtracting each scheme's mean_X row from another, as above:
# (A - Control) - (B - Control) = A - B.
intensity_models <- list(
  total_weight=m_total_cultivar,
  healthy_weight=m_healthy_cultivar,
  healthy_count_pct=m_count_pct,
  healthy_weight_pct=m_weight_pct
)
intensity_export <- list()

for (outcome in names(intensity_models)) {
  fit <- intensity_models[[outcome]]
  is_weight <- outcome %in% c("total_weight", "healthy_weight")
  groups <- if (is_weight) levels(data_clean$cultivar) else "Both cultivars"
  for (g in groups) {
    result <- intensity_results(fit, if (is_weight) g else NULL)
    cat("\n\n### ", outcome, ": ", g, "\n\n", sep="")
    # A mean-vs-zero p value is not relevant here, so omit it.
    print(knitr::kable(result$means[c("group","estimate","SE","lower","upper")],
      digits=2, caption="Adjusted means and pointwise 95% confidence intervals"))
    cat("\n\n")
    print(knitr::kable(result$comparisons, digits=4,
      caption="A-D comparisons; negative estimates mean the first group is lower"))
    cat("\n\n")

    # Order the plot by pulse count: Control, D, C, B, A.
    z <- result$means[match(c("Control","D","C","B","A"),result$means$group), ]
    plot(0:4, z$estimate, type="b", pch=19, xaxt="n",
      ylim=range(z$lower,z$upper), xlab="Heat pulses",
      ylab=if (is_weight) "Adjusted weight (g)" else "Adjusted healthy percentage (%)",
      main=paste(outcome,g))
    axis(1, at=0:4, labels=c("Control","D: 1","C: 2","B: 3","A: 4"))
    arrows(0:4,z$lower,0:4,z$upper,angle=90,code=3,length=.05)
    # Lines guide the eye; the model treats pulse schemes as categories.
    intensity_export[[paste(outcome,g)]] <- list(means=result$means,
                                                comparisons=result$comparisons)
  }
}
